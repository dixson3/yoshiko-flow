18: // stepTypeToIssueType converts a formula step type string to a types.IssueType.
19: // Returns types.TypeTask for empty types. Non-empty types pass through
20: // (trimmed and normalized) rather than being validated here: at pour and
21: // cook --persist time, flattenUnregisteredIssueTypes degrades types that
22: // are neither built-in nor registered in types.custom to task (with a
23: // warning), and the storage layer validates what remains — the same
24: // division of labor as bd create --type.
25: func stepTypeToIssueType(stepType string) types.IssueType {
26: 	stepType = strings.TrimSpace(stepType)
27: 	if stepType == "" {
28: 		return types.TypeTask
29: 	}
30: 	return types.IssueType(stepType).Normalize()
31: }
